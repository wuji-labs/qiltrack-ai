### 修复建议

```typescript
// 1. Redis故障备用限流器
export async function checkRateLimit(identifier: string, ratelimit: Ratelimit | null) {
  if (!ratelimit) {
    // ❌ 当前: 故障开放
    return { success: true, ... };

    // ✅ 修复: 使用内存限流器
    return memoryRateLimiter.limit(identifier);
  }
  // ...
}

// 2. IP验证
const TRUSTED_PROXIES = ['cloudflare', 'vercel'];
export function getIpAddress(request: Request): string {
  const forwarded = headers.get('x-forwarded-for');
  if (forwarded && isTrustedProxy(request)) {
    return forwarded.split(',')[0].trim();
  }
  return headers.get('x-real-ip') || 'unknown';
}

// 3. Symbol正则验证
const symbol = searchParams.get("symbol")?.toUpperCase().trim();
if (!symbol || !/^[A-Z0-9]{1,10}$/.test(symbol)) {
  throw new ValidationError('Invalid symbol format');
}
```
